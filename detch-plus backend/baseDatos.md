-- Trigger para actualizar stock después de insertar detalle de factura
DELIMITER //
CREATE TRIGGER tr_actualizar_stock_venta
    AFTER INSERT ON DETALLE_FACTURA
    FOR EACH ROW
BEGIN
    UPDATE STOCK 
    SET CANTIDAD_DISPONIBLE = CANTIDAD_DISPONIBLE - NEW.CAN_PRO,
        FECHA_ACTUALIZACION = NOW()
    WHERE ID_PRO = NEW.ID_PRO;
    
    -- Insertar movimiento de inventario
    INSERT INTO MOVIMIENTOS_INVENTARIO (ID_PRO, TIPO_MOV, CANTIDAD, MOTIVO, NUM_DOCUMENTO)
    VALUES (NEW.ID_PRO, 'SALIDA', NEW.CAN_PRO, 'Venta', NEW.NUM_FAC);
END //
DELIMITER ;

-- Trigger para calcular total de factura
DELIMITER //
CREATE TRIGGER tr_calcular_total_factura
    AFTER INSERT ON DETALLE_FACTURA
    FOR EACH ROW
BEGIN
    DECLARE subtotal DECIMAL(10,2);
    DECLARE iva DECIMAL(10,2);
    DECLARE total DECIMAL(10,2);
    
    SELECT SUM(PRE_TOT) INTO subtotal FROM DETALLE_FACTURA WHERE NUM_FAC = NEW.NUM_FAC;
    SET iva = subtotal * 0.15; -- IVA del 15%
    SET total = subtotal + iva;
    
    UPDATE FACTURAS 
    SET SUBTOTAL = subtotal, IVA = iva, TOTAL = total
    WHERE NUM_FAC = NEW.NUM_FAC;
END //
DELIMITER ;

        FECHA_ACTUALIZACION = NOW()
    WHERE ID_PRO = NEW.ID_PRO;
END //

END //

-- Procedimiento para agregar al carrito
DELIMITER //
CREATE PROCEDURE sp_agregar_carrito(
    IN p_id_usuario INT,
    IN p_id_producto INT,
    IN p_cantidad INT
)
BEGIN
    DECLARE v_precio DECIMAL(10,2);
    DECLARE v_stock INT;
    
    -- Obtener precio actual del producto
    SELECT VALOR INTO v_precio 
    FROM PRECIOS 
    WHERE ID_PRO = p_id_producto 
        AND TIPO_PRECIO = 'VENTA' 
        AND ACTIVO = TRUE 
    LIMIT 1;
    
    -- Verificar stock disponible
    SELECT CANTIDAD_DISPONIBLE INTO v_stock
    FROM STOCK
    WHERE ID_PRO = p_id_producto;
    
    -- Verificar si hay stock suficiente
    IF v_stock >= p_cantidad THEN
        -- Insertar o actualizar en carrito
        INSERT INTO CARRITO_COMPRAS (ID_USUARIO, ID_PRO, CANTIDAD, PRECIO_UNITARIO)
        VALUES (p_id_usuario, p_id_producto, p_cantidad, v_precio)
        ON DUPLICATE KEY UPDATE 
            CANTIDAD = CANTIDAD + p_cantidad,
            PRECIO_UNITARIO = v_precio,
            FECHA_ACTUALIZACION = NOW();
            
        SELECT 'SUCCESS' as RESULTADO, 'Producto agregado al carrito' as MENSAJE;
    ELSE
        SELECT 'ERROR' as RESULTADO, 'Stock insuficiente' as MENSAJE;
    END IF;
END //
DELIMITER ;

-- Procedimiento para procesar pedido
DELIMITER //
CREATE PROCEDURE sp_procesar_pedido(
    IN p_id_usuario INT,
    IN p_id_direccion_envio INT,
    IN p_metodo_pago VARCHAR(50),
    IN p_codigo_cupon VARCHAR(20)
)
BEGIN
    DECLARE v_id_pedido INT;
    DECLARE v_subtotal DECIMAL(10,2) DEFAULT 0;
    DECLARE v_descuento DECIMAL(10,2) DEFAULT 0;
    DECLARE v_impuestos DECIMAL(10,2) DEFAULT 0;
    DECLARE v_total DECIMAL(10,2) DEFAULT 0;
    DECLARE done INT DEFAULT FALSE;
    
    DECLARE carrito_cursor CURSOR FOR 
        SELECT ID_PRO, CANTIDAD, PRECIO_UNITARIO 
        FROM CARRITO_COMPRAS 
        WHERE ID_USUARIO = p_id_usuario;
    
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
    
    -- Calcular subtotal del carrito
    SELECT SUM(CANTIDAD * PRECIO_UNITARIO) INTO v_subtotal
    FROM CARRITO_COMPRAS
    WHERE ID_USUARIO = p_id_usuario;
    
    -- Aplicar descuento si hay cupón
    IF p_codigo_cupon IS NOT NULL THEN
        SELECT 
            CASE 
                WHEN TIPO_DESCUENTO = 'PORCENTAJE' 
                THEN v_subtotal * (VALOR_DESCUENTO / 100)
                ELSE VALOR_DESCUENTO
            END INTO v_descuento
        FROM CUPONES
        WHERE CODIGO_CUPON = p_codigo_cupon
            AND ACTIVO = TRUE
            AND NOW() BETWEEN FECHA_INICIO AND FECHA_FIN
            AND v_subtotal >= MONTO_MINIMO
            AND (LIMITE_USO IS NULL OR USOS_ACTUALES < LIMITE_USO);
    END IF;
    
    SET v_impuestos = (v_subtotal - v_descuento) * 0.15; -- IVA 15%
    SET v_total = v_subtotal - v_descuento + v_impuestos;
    
    -- Crear pedido
    INSERT INTO PEDIDOS_ONLINE (
        ID_USUARIO, ID_DIRECCION_ENVIO, SUBTOTAL, IMPUESTOS, 
        DESCUENTO, TOTAL, METODO_PAGO
    ) VALUES (
        p_id_usuario, p_id_direccion_envio, v_subtotal, v_impuestos,
        v_descuento, v_total, p_metodo_pago
    );
    
    SET v_id_pedido = LAST_INSERT_ID();
    
    -- Transferir items del carrito al pedido
    INSERT INTO DETALLE_PEDIDO_ONLINE (ID_PEDIDO, ID_PRO, CANTIDAD, PRECIO_UNITARIO, PRECIO_TOTAL)
    SELECT v_id_pedido, ID_PRO, CANTIDAD, PRECIO_UNITARIO, (CANTIDAD * PRECIO_UNITARIO)
    FROM CARRITO_COMPRAS
    WHERE ID_USUARIO = p_id_usuario;
    
    -- Actualizar uso de cupón si aplica
    IF p_codigo_cupon IS NOT NULL AND v_descuento > 0 THEN
        UPDATE CUPONES SET USOS_ACTUALES = USOS_ACTUALES + 1 
        WHERE CODIGO_CUPON = p_codigo_cupon;
        
        INSERT INTO CUPONES_UTILIZADOS (ID_CUPON, ID_USUARIO, ID_PEDIDO, MONTO_DESCUENTO)
        SELECT ID_CUPON, p_id_usuario, v_id_pedido, v_descuento
        FROM CUPONES WHERE CODIGO_CUPON = p_codigo_cupon;
    END IF;
    
    SELECT v_id_pedido as ID_PEDIDO, v_total as TOTAL;
END //
DELIMITER ;
